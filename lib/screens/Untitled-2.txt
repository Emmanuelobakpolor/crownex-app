/// Deposit NGN into the wallet via the Flutterwave standard checkout.
///
/// Flow: enter amount → POST /wallet/initiate-deposit/ for a public key →
/// launch the Flutterwave SDK → on SDK success, POST /wallet/verify-payment/
/// (the SDK result alone never credits the wallet — only a verified server
/// call does) → show the new balance.
class DepositNgnScreen extends StatefulWidget {
  const DepositNgnScreen({super.key});

  static const _appPrefix = 'CROWNEX';

  @override
  State<DepositNgnScreen> createState() => _DepositNgnScreenState();
}

class _DepositNgnScreenState extends State<DepositNgnScreen> {
  final _amountController = TextEditingController();
  final _amountFocus = FocusNode();
  final _wallet = WalletService.instance;
  final _random = Random.secure();

  bool _submitting = false;
  String? _loadingLabel;

  static const _quickAmounts = [500, 1000, 2000, 5000, 10000, 20000];

  @override
  void dispose() {
    _amountController.dispose();
    _amountFocus.dispose();
    super.dispose();
  }

  double? get _amount {
    final raw = _amountController.text.replaceAll(',', '').trim();
    if (raw.isEmpty) return null;
    return double.tryParse(raw);
  }

  bool get _canPay {
    final amount = _amount;
    return amount != null && amount >= 100 && !_submitting;
  }

  String? get _amountHint {
    final amount = _amount;
    if (amount == null) return null;
    if (amount < 100) return 'Minimum deposit is ₦100';
    return null;
  }

  void _selectQuickAmount(int value) {
    _amountController.text = value.toString();
    setState(() {});
    _amountFocus.unfocus();
  }

  String _generateTxRef() {
    final millis = DateTime.now().millisecondsSinceEpoch;
    final suffix = (_random.nextInt(900000) + 100000).toString();
    return '${DepositNgnScreen._appPrefix}-$millis-$suffix';
  }

  Future<void> _pay() async {
    if (!_canPay) return;
    final amount = _amount!;

    setState(() {
      _submitting = true;
      _loadingLabel = 'Preparing payment…';
    });

    try {
      final txRef = _generateTxRef();
      final init = await _wallet.initiateDeposit(amount: amount, txRef: txRef);

      if (!mounted) return;
      setState(() => _loadingLabel = null);

      final isTestMode = init.publicKey.contains('TEST');
      final flutterwave = Flutterwave(
        publicKey: init.publicKey,
        currency: 'NGN',
        redirectUrl: 'https://flutterwave.com',
        txRef: init.txRef,
        amount: init.amount.toStringAsFixed(2),
        customer: Customer(
          name: init.customerName,
          phoneNumber: init.customerPhone,
          email: init.customerEmail,
        ),
        paymentOptions: 'card,banktransfer,ussd',
        customization: Customization(
          title: 'CrownEx Top-Up',
          description: 'Add money to wallet',
        ),
        isTestMode: isTestMode,
      );

      final response = await flutterwave.charge(context);
      if (!mounted) return;

      if (response.success != true) {
        // The SDK result alone never touches a transaction record — only a
        // verify-payment call does. Without this, a cancelled/failed charge
        // leaves its transaction stuck as "Pending" forever instead of
        // resolving to "Failed", so we still ask the server to check.
        try {
          await _wallet.verifyPayment(init.txRef);
        } catch (_) {
          // Expected: verification fails for a cancelled/failed charge,
          // which is what flips the transaction to Failed server-side.
        }

        if (!mounted) return;
        setState(() {
          _submitting = false;
          _loadingLabel = null;
        });
        _showSnack(
          response.status == 'cancelled'
              ? 'Payment cancelled.'
              : 'Payment failed. Please try again.',
        );
        return;
      }

      setState(() => _loadingLabel = 'Verifying payment…');
      final newBalance = await _wallet.verifyPayment(init.txRef);

      if (!mounted) return;
      setState(() {
        _submitting = false;
        _loadingLabel = null;
      });
      await _showSuccessDialog(amount: amount, newBalance: newBalance);
      if (mounted) Navigator.of(context).pop(true);
    } on ApiException catch (e) {
      if (!mounted) return;
      setState(() {
        _submitting = false;
        _loadingLabel = null;
      });
      _showSnack(e.message);
    } catch (_) {
      if (!mounted) return;
      setState(() {
        _submitting = false;
        _loadingLabel = null;
      });
      _showSnack('Something went wrong. Please try again.');
    }
  }

  void _showSnack(String message) {
    ScaffoldMessenger.of(context).hideCurrentSnackBar();
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text(message),
        behavior: SnackBarBehavior.floating,
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(12)),
        margin: const EdgeInsets.fromLTRB(16, 0, 16, 16),
      ),
    );
  }

  Future<void> _showSuccessDialog({
    required double amount,
    required double newBalance,
  }) {
    return showDialog<void>(
      context: context,
      barrierDismissible: false,
      builder: (ctx) => _DepositSuccessDialog(
        amount: amount,
        newBalance: newBalance,
        onDone: () => Navigator.of(ctx).pop(),
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    final bottomPad = MediaQuery.paddingOf(context).bottom;
    final amount = _amount;
    final amountHint = _amountHint;
    final selectedQuick = amount != null &&
            amount == amount.roundToDouble() &&
            _quickAmounts.contains(amount.round())
        ? amount.round()
        : null;

    return AnnotatedRegion<SystemUiOverlayStyle>(
      value: SystemUiOverlayStyle.dark.copyWith(
        statusBarColor: Colors.transparent,
      ),
      child: Scaffold(
        backgroundColor: AppColors.background,
        body: SafeArea(
          child: Stack(
            children: [
              Column(
                children: [
                  const _DepositHeader(),
                  Expanded(
                    child: SingleChildScrollView(
                      padding: const EdgeInsets.fromLTRB(20, 4, 20, 24),
                      physics: const BouncingScrollPhysics(),
                      child: Column(
                        crossAxisAlignment: CrossAxisAlignment.stretch,
                        children: [
                          _AmountCard(
                            controller: _amountController,
                            focusNode: _amountFocus,
                            amountHint: amountHint,
                            onChanged: (_) => setState(() {}),
                          ),
                          const SizedBox(height: 16),
                          _QuickAmountRow(
                            amounts: _quickAmounts,
                            selected: selectedQuick,
                            onSelected: _selectQuickAmount,
                          ),
                          const SizedBox(height: 28),
                          const _SectionLabel('Payment methods'),
                          const SizedBox(height: 12),
                          const _PaymentMethodsGrid(),
                          const SizedBox(height: 20),
                          const _SecureInfoBanner(),
                        ],
                      ),
                    ),
                  ),
                  _BottomBar(
                    bottomPad: bottomPad,
                    canPay: _canPay,
                    submitting: _submitting && _loadingLabel != null,
                    amount: amount,
                    onPay: _pay,
                  ),
                ],
              ),
              if (_loadingLabel != null)
                _LoadingOverlay(label: _loadingLabel!),
            ],
          ),
        ),
      ),
    );
  }
}

// ─── Quick amounts ──────────────────────────────────────────────────────────

class _QuickAmountRow extends StatelessWidget {
  const _QuickAmountRow({
    required this.amounts,
    required this.selected,
    required this.onSelected,
  });

  final List<int> amounts;
  final int? selected;
  final ValueChanged<int> onSelected;

  String _label(int value) {
    if (value >= 1000) {
      final k = value / 1000;
      return k == k.roundToDouble()
          ? '₦${k.toInt()}k'
          : '₦${k.toStringAsFixed(1)}k';
    }
    return '₦$value';
  }

  @override
  Widget build(BuildContext context) {
    return Wrap(
      spacing: 8,
      runSpacing: 8,
      children: amounts.map((amount) {
        final isSelected = selected == amount;
        return Material(
          color: Colors.transparent,
          child: InkWell(
            onTap: () => onSelected(amount),
            borderRadius: BorderRadius.circular(12),
            child: AnimatedContainer(
              duration: const Duration(milliseconds: 160),
              padding: const EdgeInsets.symmetric(horizontal: 14, vertical: 10),
              decoration: BoxDecoration(
                color: isSelected
                    ? AppColors.primary
                    : AppColors.surface,
                borderRadius: BorderRadius.circular(12),
                border: Border.all(
                  color: isSelected
                      ? AppColors.primary
                      : AppColors.border,
                  width: 1.2,
                ),
                boxShadow: isSelected
                    ? [
                        BoxShadow(
                          color: AppColors.primary.withValues(alpha: 0.22),
                          blurRadius: 10,
                          offset: const Offset(0, 4),
                        ),
                      ]
                    : null,
              ),
              child: Text(
                _label(amount),
                style: TextStyle(
                  fontSize: 13.5,
                  fontWeight: FontWeight.w600,
                  color: isSelected ? Colors.white : AppColors.textPrimary,
                ),
              ),
            ),
          ),
        );
      }).toList(),
    );
  }
}

// ─── Bottom bar ─────────────────────────────────────────────────────────────

class _BottomBar extends StatelessWidget {
  const _BottomBar({
    required this.bottomPad,
    required this.canPay,
    required this.submitting,
    required this.amount,
    required this.onPay,
  });

  final double bottomPad;
  final bool canPay;
  final bool submitting;
  final double? amount;
  final VoidCallback onPay;

  String _format(double v) {
    final s = v.toStringAsFixed(v.truncateToDouble() == v ? 0 : 2);
    final parts = s.split('.');
    final whole = parts[0];
    final buf = StringBuffer();
    for (var i = 0; i < whole.length; i++) {
      if (i > 0 && (whole.length - i) % 3 == 0) buf.write(',');
      buf.write(whole[i]);
    }
    if (parts.length > 1) return '${buf.toString()}.${parts[1]}';
    return buf.toString();
  }

  @override
  Widget build(BuildContext context) {
    final label = amount != null && canPay
        ? 'Continue · ₦${_format(amount!)}'
        : 'Continue to payment';

    return Container(
      padding: EdgeInsets.fromLTRB(20, 12, 20, 12 + bottomPad),
      decoration: BoxDecoration(
        color: AppColors.surface,
        border: Border(
          top: BorderSide(color: AppColors.border.withValues(alpha: 0.8)),
        ),
        boxShadow: [
          BoxShadow(
            color: Colors.black.withValues(alpha: 0.04),
            blurRadius: 16,
            offset: const Offset(0, -4),
          ),
        ],
      ),
      child: Column(
        mainAxisSize: MainAxisSize.min,
        children: [
          AuthPrimaryButton(
            label: label,
            enabled: canPay,
            loading: submitting,
            onPressed: onPay,
          ),
          const SizedBox(height: 10),
          const Row(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              Icon(Icons.lock_outline_rounded, size: 13, color: AppColors.textMuted),
              SizedBox(width: 5),
              Text(
                'Secured by Flutterwave',
                style: TextStyle(
                  fontSize: 12,
                  fontWeight: FontWeight.w500,
                  color: AppColors.textMuted,
                ),
              ),
            ],
          ),
        ],
      ),
    );
  }
}

    for (var i = 0; i < whole.length; i++) {
      if (i > 0 && (whole.length - i) % 3 == 0) buf.write(',');
      buf.write(whole[i]);
    }
    return '${buf.toString()}.${parts[1]}';
  }
}
